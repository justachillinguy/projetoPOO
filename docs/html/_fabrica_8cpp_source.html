551int Fabrica::Aviso_Humidade(list<Motor*>& lm)
552{
553 lm.clear();
554
555 for (list<Sensor*>::iterator its = sensores.begin(); its != sensores.end(); ++its)
556 {
557 Sensor* s = *its;
558 if (s->GetTipo() != "HUMIDADE" || !s->EmAviso())
559 {
560 continue;
561 }
562
563 int raio = s->GetRaioVizinhanca();
564
565 for (list<Motor*>::iterator itm = motores.begin(); itm != motores.end(); ++itm)
566 {
567 Motor* m = *itm;
568 int dx = m->GetX() - s->GetX();
569 int dy = m->GetY() - s->GetY();
570 double dist = sqrt((double)(dx * dx + dy * dy));
571
572 bool jaNaLista = false;
573 for (list<Motor*>::iterator ild = lm.begin(); ild != lm.end(); ++ild)
574 {
575 if (*ild == m) { jaNaLista = true; break; }
576 }
577
578 if (dist <= raio && m->GetEstado() != STOP && m->GetEstado() != AVARIADO && !jaNaLista)
579 {
580 m->STOP();
581 lm.push_back(m);
582 }
583 }
584 }

586 if (!lm.empty())

588 cout << "[AVISO GESTOR] Sensor de HUMIDADE em aviso! " << lm.size() << " motor(es) desligado(s)." << endl;
